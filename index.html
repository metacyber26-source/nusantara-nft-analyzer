import { GoogleGenAI, Type } from '@google/genai'; import fs from 'fs'; const ai = new GoogleGenAI({ apiKey: process.env.GEMINI_API_KEY }); // Fungsi untuk konversi gambar lokal ke format inlineData function fileToGenerativePart(path, mimeType) { return { inlineData: { data: Buffer.from(fs.readFileSync(path)).toString("base64"), mimeType }, }; } // Schema agar output AI selalu terstruktur dan obyektif const analysisSchema = { type: Type.OBJECT, properties: { visualDescription: { type: Type.STRING, description: "Deskripsi obyektif visual secara keseluruhan." }, elementsAnalysis: { type: Type.ARRAY, items: { type: Type.OBJECT, properties: { part: { type: Type.STRING, description: "Bagian gambar: body, face, tail, eyes, eyebrow, nose, ears, beard, atau background" }, attributesFound: { type: Type.OBJECT, properties: { shape: { type: Type.STRING, description: "Bentuk yang terdeteksi secara fisik pada gambar" }, motif: { type: Type.STRING, description: "Motif atau tekstur yang terlihat" }, color: { type: Type.STRING, description: "Warna dominan atau aksen" } }, required: ["shape", "motif", "color"] }, philosophicalMeaning: { type: Type.STRING, description: "Arti filosofis berdasarkan bentuk, motif, dan warna spesifik bagian ini." }, fengShuiAnalysis: { type: Type.STRING, description: "Analisis fengshui (elemen 5 unsur, aliran qi, simbolisme keberuntungan/keseimbangan)." } }, required: ["part", "attributesFound", "philosophicalMeaning", "fengShuiAnalysis"] } } }, required: ["visualDescription", "elementsAnalysis"] }; async function analyzeImage(imagePath, mimeType) { const imagePart = fileToGenerativePart(imagePath, mimeType); const prompt = ` Bertindaklah sebagai ahli semiotika, filosofi visual, dan praktisi Fengshui profesional. Tugasmu adalah menganalisis GAMBAR INI SECARA OBYEKTIF berdasarkan apa yang benar-benar terlihat. Analisis elemen-elemen berikut jika ada pada gambar: 1. Body (Tubuh) 2. Face (Wajah) 3. Tail (Ekor) 4. Eyes (Mata) 5. Eyebrow (Alis) 6. Nose (Hidung) 7. Ears (Telinga) 8. Beard (Janggut/Kumis) 9. Background (Latar Belakang) Untuk setiap elemen yang ditemukan: - Identifikasi spesifik **Bentuk (Shape)**, **Motif/Tekstur (Motif)**, dan **Warna (Color)** yang nyata terlihat pada gambar. - Berikan **Arti Filosofis** yang relevan dengan bentuk, motif, dan warna tersebut. - Berikan **Analisis Fengshui** (misal: keterkaitan dengan 5 Elemen Wood/Fire/Earth/Metal/Water, simbol rezeki, proteksi, atau keseimbangan energi). Catatan Penting: - Beda visual wajib menghasilkan beda interpretasi. - Jika suatu elemen tidak ada di dalam gambar (misal: tidak ada ekor/tail), abaikan elemen tersebut. `; try { const response = await ai.models.generateContent({ model: 'gemini-2.5-flash', contents: [prompt, imagePart], config: { responseMimeType: 'application/json', responseSchema: analysisSchema, temperature: 0.2, // Temperature rendah agar hasil konsisten dan obyektif } }); const result = JSON.parse(response.text); return result; } catch (error) { console.error("Gagal menganalisis gambar:", error); } } // Cara Penggunaan analyzeImage("./sampel_gambar.png", "image/png").then((data) => { console.log(JSON.stringify(data, null, 2)); });
